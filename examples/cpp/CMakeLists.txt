cmake_minimum_required(VERSION 3.16)
project(tdlib_json_example LANGUAGES CXX)
option(TDLIB_EXAMPLE_STATIC "Link TDLib's static JSON API instead of its shared library" OFF)
find_package(Td REQUIRED CONFIG)
add_executable(tdlib-cpp main.cpp)
target_compile_features(tdlib-cpp PRIVATE cxx_std_17)
if(TDLIB_EXAMPLE_STATIC)
  target_link_libraries(tdlib-cpp PRIVATE Td::TdJsonStatic)
else()
  target_link_libraries(tdlib-cpp PRIVATE Td::TdJson)
endif()
